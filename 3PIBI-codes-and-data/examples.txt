## Data-driven case example: detecting nonlocality on a pure qutrit Dicke state
% start by generating the qutrit pure Dicke state for n=15 parties
n=32; d=3; dim=nchoosek(n+d-1, d-1);
dickePure=zeros(dim, 1); % to initialize the state in the symmetric subspace of dimensions nchoosek(n+d-1, d-1)
dickePure(340) = 1; % quite balanced in excitations between the three-levels

% next we obtaining its 2-body RDM and the projectors of the measurements to later get the vector of conditional probabilities p
load('A_matrices.mat', 'A32');
rho2 = get_rdm(dickePure*dickePure',3,2,A32);

% and the projectors of the measurements in two directions of the Sx-Sz plane that we arbitrarily choose
[Sx,Sy,Sz,Qyz,Qxz,Qxy,Qxx,Qzz,Qyy] = generateS(); % call script to get the dipole S and quadrupole Q observables
t=43;q=50;step=pi/100; % set the arbitrary directions
proj0 = get_proj(Sx*cos(t*step)+Sz*sin(t*step)); % obtain the projectors from the observables
proj1 = get_proj(Sx*cos(q*step)+Sz*sin(q*step));

% finally we get the vector of conditional probabilities
p = build_point_fulldim(n, rho2, proj0, proj1);

% Now we proceed to call the data-driven SDP method
load('M_constraints.mat', 'M32');
sdp_res = build_sdp_spin1_violation_nolambda(M32,p); % which is infeasible, thus showing that p is outside of the local polytope
BI = sdp_res{2}; % here we have stored the dual variables of the infeasible problem, which are a Bell inequality violated by p:
BI*p' % to see that indeed there is Bell violation
